```python
"""Seat-level permission enforcement for dsh-council plugin.

Loaded from council_config.yaml -> permissions.defaults and permissions.seats[seat_id]."""

import fnmatch
import re
from typing import Dict, List, Optional, Set

class PermissionError(Exception):
    pass

class SeatPermissions:
    def __init__(self, seat_id: str, profile: dict, defaults: dict):
        self.seat_id = seat_id
        # merge profile over defaults
        merged = defaults.copy()
        merged.update(profile)
        self.rules: Dict[str, dict] = merged

    def check(self, tool_name: str, **kwargs) -> bool:
        """Return True if permitted, raise PermissionError otherwise."""
        rule = self.rules.get(tool_name)
        if rule is None:   # undefined tool → deny
            raise PermissionError(f"Seat '{self.seat_id}' unknown tool '{tool_name}' denied")
        action = rule.get("action", "deny")
        if action == "deny":
            raise PermissionError(f"Seat '{self.seat_id}' tool '{tool_name}' denied by explicit rule")
        # allow, with optional further filtering
        if tool_name in ("tools.file_read", "tools.file_write", "tools.file_exec"):
            allowed_paths = rule.get("paths", [])
            target_path = kwargs.get("path", "")
            if not any(fnmatch.fnmatch(target_path, p) for p in allowed_paths):
                raise PermissionError(f"Seat '{self.seat_id}' tool '{tool_name}' path '{target_path}' not in allowed paths")
        if tool_name == "tools.network.http":
            allowed_domains = rule.get("domains", [])
            target_host = kwargs.get("host", "")
            if target_host not in allowed_domains:
                raise PermissionError(f"Seat '{self.seat_id}' tool '{tool_name}' host '{target_host}' not in allowed domains")
        if tool_name == "tools.codex.model":
            allowed_models = rule.get("models", [])
            target_model = kwargs.get("model", "")
            if target_model not in allowed_models:
                raise PermissionError(f"Seat '{self.seat_id}' tool '{tool_name}' model '{target_model}' not in allowed models")
        return True
```

import path from 'node:path';

export interface CapabilityEnvelope {
  runId: string;
  agentId: string;
  planId: string;
  expiresAt: number;
  readableRoots: string[];
  writableRoot: string;
  allowedCommands: string[];
  networkPolicy: 'loopback' | 'hosted-apis-only' | 'blocked';
  protectedPaths: string[];
}

export class PermissionEnforcer {
  constructor(private envelope: CapabilityEnvelope) {}

  public validatePathRead(targetPath: string): boolean {
    if (Date.now() > this.envelope.expiresAt) return false;
    const absPath = path.resolve(targetPath);
    for (const protectedPath of this.envelope.protectedPaths) {
      if (absPath.startsWith(path.resolve(protectedPath))) {
        return false;
      }
    }
    return this.envelope.readableRoots.some(root => absPath.startsWith(path.resolve(root)));
  }

  public validatePathWrite(targetPath: string): boolean {
    if (Date.now() > this.envelope.expiresAt) return false;
    const absPath = path.resolve(targetPath);
    for (const protectedPath of this.envelope.protectedPaths) {
      if (absPath.startsWith(path.resolve(protectedPath))) {
        return false;
      }
    }
    return absPath.startsWith(path.resolve(this.envelope.writableRoot));
  }

  public validateCommand(command: string): boolean {
    if (Date.now() > this.envelope.expiresAt) return false;
    const baseCmd = command.trim().split(/\s+/)[0];
    if (baseCmd === 'git' && command.includes('push')) {
      return false; // Pushes strictly forbidden
    }
    return this.envelope.allowedCommands.some(allowed => baseCmd === allowed || command.startsWith(allowed));
  }
}

import os
from dataclasses import dataclass, field
from typing import List

@dataclass
class PermissionEnvelope:
    worker_id: str
    read_roots: List[str]
    write_root: str
    allowed_commands: List[str]
    allow_network: bool = False
    
    def _is_subpath(self, parent: str, child: str) -> bool:
        parent_abs = os.path.abspath(parent)
        child_abs = os.path.abspath(child)
        return os.path.commonpath([parent_abs]) == os.path.commonpath([parent_abs, child_abs])

    def can_read(self, path: str) -> bool:
        return any(self._is_subpath(root, path) for root in self.read_roots)
        
    def can_write(self, path: str) -> bool:
        if not self.write_root:
            return False
        return self._is_subpath(self.write_root, path)

    def can_execute(self, command: str) -> bool:
        if not command.strip():
            return False
        cmd_base = command.split()[0].lower()
        if cmd_base.endswith('.exe') or cmd_base.endswith('.cmd'):
            cmd_base = os.path.splitext(cmd_base)[0]
        return cmd_base in [c.lower() for c in self.allowed_commands]

import subprocess
import os

class WorktreeManager:
    def __init__(self, base_repo_path: str, worktree_dir: str = ".dsh/worktrees"):
        self.base_repo_path = os.path.abspath(base_repo_path)
        self.worktree_dir = os.path.join(self.base_repo_path, worktree_dir)
        os.makedirs(self.worktree_dir, exist_ok=True)

    def provision_worker_tree(self, worker_id: str) -> str:
        target_path = os.path.join(self.worktree_dir, worker_id)
        if not os.path.exists(target_path):
            subprocess.run(
                ["git", "worktree", "add", "-b", f"dsh-worker-{worker_id}", target_path],
                cwd=self.base_repo_path,
                check=True,
                capture_output=True
            )
        return target_path

    def cleanup_worker_tree(self, worker_id: str):
        target_path = os.path.join(self.worktree_dir, worker_id)
        if os.path.exists(target_path):
            subprocess.run(["git", "worktree", "remove", "-f", target_path], cwd=self.base_repo_path, check=True)
            subprocess.run(["git", "branch", "-D", f"dsh-worker-{worker_id}"], cwd=self.base_repo_path, check=True)
